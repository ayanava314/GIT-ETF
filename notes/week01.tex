\section{Motivation:}

Let $ V $ be a $ K $ vector space where $ K $ is algebraically closed.

\begin{definition}
	Let $ Fr_n(V) := \{(v_i)_{i=1}^{n}\in V^n \mid v_i\ne \lambda v_j, $ for  $i\ne j, $  $span(\{v_i\})=V \} $ be called frame.
\end{definition} 

We define some actions of different groups on $ Fr_n(V)\times Fr_n(V^*) $:

\begin{itemize}
	\item $ GL(V) $ acts on the left as : $ M((v_i),(\phi_i)) = ((Mv_i),(v \mapsto \phi_i(M^{-1}v))) $
	\item $ (K^\times)^n\rtimes S_n $ acts on the right as (scaling and permuting frames): \begin{itemize}
		\item $ ((v_i),(\phi_i))\lambda_i = ((\lambda_i v_i), (v\mapsto\phi_i(\lambda_i^{-1}v)))$
		\item $ ((v_i),(\phi_i))\sigma= ((v_{\sigma(i)})(\phi_{\sigma(i)})) $
	\end{itemize} 
\end{itemize}

\begin{definition}
	Now we need to look at the subspace of $ GL(V) \backslash Fr_n(V)\times Fr_n(V^*) / (K^\times)^n\rtimes S_n $ called {$ \textbf{ETB}_\textbf{n}\textbf{(V)} $} which satisfies the following:
	\begin{enumerate}
		\item $ \phi_i(v_i)=1 $
		\item $ \phi_i(v_j)\phi_j(v_i) =\frac{n-d}{d(n-1)} $ for $ i\ne j $
		\item $ v_j = \frac{d}{n} \sum\limits_{i=1}^{n} \phi_i(v_j)v_i $
	\end{enumerate}
\end{definition}

\vspace{0.2in}
$\underline{\text{Now the \textbf{BIG QUESTIONS} are:}}$ 
\begin{itemize}
	\item what is $ ETB_n(V) $? Variety? 
	\item when is dimension of $ ETB_n(V)$ zero? In that case, what is the number of points $ \# ETB_n(V) $?
\end{itemize}

$\underline{\textit{Special Case:}}$\\

$ K = \mathbb{C}, \phi_i(v) = \langle v_i,v \rangle= \bar{v_i},v $ is called $ \textbf{ETF} $ Equiangular Tight Frames...
\begin{enumerate}
	\item unit norm: $ \langle v_i,v_i\rangle =1 $
	\item equiangular: $ \mid \langle v_i,v_j\rangle \mid ^2 = \dfrac{n-d}{d(n-1)}  $ 
	\item tight: $ v = \frac{d}{n}\sum\limits_{i=1}^n \langle v_i,v \rangle v_i $
\end{enumerate}
